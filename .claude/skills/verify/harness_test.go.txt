package verify

import (
	"context"
	"os"
	"testing"
	"time"

	"ride-home-router/internal/access/accesstest"
	"ride-home-router/internal/routefeedback"
	"ride-home-router/internal/server"
)

// TestServe runs the real server (server.New: real mux, handlers, templates,
// Postgres) on 127.0.0.1:8099 with a synthetic Clerk backend, writes an admin
// bearer token to /tmp/verify-token, and stops when /tmp/verify-stop exists.
func TestServe(t *testing.T) {
	fixture := accesstest.New(t)
	token := fixture.Admin()
	fixture.User("user_member", []string{"member@example.test"}, nil)
	fixture.Session("sess_member", "user_member", "active")
	memberToken := fixture.Token("user_member", "sess_member")
	routefeedback.SetTrustCFAccessHeader(os.Getenv("TRUST_CF_ACCESS_HEADER") == "true")
	srv, err := server.New(context.Background(), server.Config{
		Auth:                    fixture.Config("admin@example.test"),
		Addr:                    "127.0.0.1:8099",
		DatabaseURL:             os.Getenv("DATABASE_URL"),
		CredentialEncryptionKey: os.Getenv("CREDENTIAL_ENCRYPTION_KEY"),
	})
	if err != nil {
		t.Fatal(err)
	}
	addr, err := srv.Start()
	if err != nil {
		t.Fatal(err)
	}
	_ = os.WriteFile("/tmp/verify-token", []byte(token), 0o600)
	_ = os.WriteFile("/tmp/verify-member-token", []byte(memberToken), 0o600)
	t.Logf("listening on %s", addr)
	for {
		if _, err := os.Stat("/tmp/verify-stop"); err == nil {
			break
		}
		select {
		case err := <-srv.Errors():
			t.Fatal(err)
		case <-time.After(500 * time.Millisecond):
		}
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_ = srv.Shutdown(ctx)
}
